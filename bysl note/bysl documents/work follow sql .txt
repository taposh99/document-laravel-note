       actor_ids, designation_ids, other_unit_approver,
       all_status_ids, ending_status_ids, rejection_status_ids
FROM process_flow_steps
WHERE NOT JSON_VALID(actor_ids)
   OR NOT JSON_VALID(designation_ids)
   OR NOT JSON_VALID(other_unit_approver)
   OR NOT JSON_VALID(all_status_ids)
   OR NOT JSON_VALID(ending_status_ids)
   OR NOT JSON_VALID(rejection_status_ids);
