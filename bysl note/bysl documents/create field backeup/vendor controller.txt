<?php

namespace AppSmartWarehouse\Transaction\Http\Controllers;

use AppSmartWarehouse\Transaction\Entities\Vendor;
use AppSmartWarehouse\Warehouse\Helpers\FileUpload;
use AppSmartWarehouse\Transaction\Services\VendorService;
use Illuminate\Contracts\Support\Renderable;
use Illuminate\Http\Request;
use Illuminate\Routing\Controller;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Response;
use Illuminate\Http\Exceptions\HttpResponseException;
use Illuminate\Database\Eloquent\ModelNotFoundException;
use Illuminate\Support\Facades\DB;
use Symfony\Component\HttpKernel\Event\ResponseEvent;
use Throwable, Exception;
  use Maatwebsite\Excel\Facades\Excel;
use App\Exports\VendorExport;

class VendorController extends Controller
{
    public function __construct(public VendorService $vendorService,public FileUpload $fileUpload)
    {

        $this->vendorService = $vendorService;
        $this->fileUpload = $fileUpload;
    }

    /**
     * Display a listing of the resource.
     * @return Renderable
     */
    public function index(Request $request): JsonResponse
    {
        $filter = $this->vendorService->filterValidation($request)->validate();    
        $datas = $this->vendorService->getList($filter); 
        return response()->json(responseBuilder(Response::HTTP_OK, "Success", $datas->toArray()), Response::HTTP_OK);
    }

    /**
     * Store a newly created resource in storage.
     * @param Request $request
     * @return Renderable
     */
public function store(Request $request)
{
     \DB::listen(function ($query) {
        \Log::info('SQL: ' . $query->sql, $query->bindings);
    });
    DB::beginTransaction();
    try {

        $vData = $this->vendorService->validation($request);
        if ($vData->fails()) {
           return response()->json(
                responseBuilder(
                    Response::HTTP_UNPROCESSABLE_ENTITY,
                    $vData->errors()->first(),
                    $vData->errors()
                ),
                Response::HTTP_UNPROCESSABLE_ENTITY
            );
        }
        $data = $vData->validate();
        
        $imagePath = null;
        if ($request->hasFile('photo')) {
            $imagePath = $this->fileUpload->uploadImage($request, 'photo');
        } 
        $result = $this->vendorService->create($data, $imagePath, $request);

        DB::commit();
        return response()->json(responseBuilder(Response::HTTP_CREATED, "Vendor created successfully.", $result->toArray()), Response::HTTP_OK);
    } catch (Throwable $exception) {
        DB::rollBack();
        // 👇 dd() hatiye proper JSON error response koro
        \Log::error('Vendor store exception: ' . $exception->getMessage());
        \Log::error($exception->getTraceAsString());
        
        return response()->json(
            responseBuilder(
                Response::HTTP_INTERNAL_SERVER_ERROR,
                'Something went wrong: ' . $exception->getMessage(),
                []
            ),
            Response::HTTP_INTERNAL_SERVER_ERROR
        );
    }
}

    /**
     * Show the specified resource.
     * @param int $id
     * @return Renderable
     */
    public function show($id)
    {
        ValidateUUID($id);
        $result = $this->vendorService->getDetails($id);
        return response()->json(
            responseBuilder(Response::HTTP_OK, "Vendor viewed successfully.", $result->toArray()),
            Response::HTTP_OK
        );
    }

    /**
     * Update the specified resource in storage.
     * @param Request $request
     * @param string $id
     * @return JsonResponse
     * @throws Throwable
     * @throws ValidationException
     */

    public function update(Request $request, string $id)
    {
        ValidateUUID($id);
        DB::beginTransaction();
        try {
            $vData = $this->vendorService->validation($request, $id);
            if ($vData->fails()) {
               return response()->json(
                    responseBuilder(
                        Response::HTTP_UNPROCESSABLE_ENTITY,
                        $vData->errors()->first(),
                        $vData->errors()
                    ),
                    Response::HTTP_UNPROCESSABLE_ENTITY
                );
            }
            $data = $vData->validate();
            $vendor = Vendor::where('id',$id)->firstOrFail();

            $imagePath=null;
            if ($request->hasFile('photo')) {

                $imagePath = $this->fileUpload->uploadImage($request,'photo','/uploads',$vendor->photo);
            } 

            $this->vendorService->update($data, $vendor, $imagePath);

            DB::commit();
            return response()->json(responseBuilder(Response::HTTP_OK, "Vendor updated successfully.", []), Response::HTTP_OK);
        } catch (Throwable $exception) {
            DB::rollBack();
            throw $exception;
        }
    }

    /**
     * Remove the specified resource from storage.
     */

    // public function destroy(Request $request): JsonResponse
    // {

    //     $this->vendorService->delete($validated);

    public function destroy(Request $request): JsonResponse
    {
        $validated = $this->vendorService->deleteValidation($request);

        $result = $this->vendorService->delete($validated);

        if (!$result['success']) {
            return response()->json(
                responseBuilder(
                    Response::HTTP_CONFLICT,
                    $result['message'],
                    []
                ),
                Response::HTTP_CONFLICT
            );
        }

        return response()->json(
            responseBuilder(
                Response::HTTP_OK,
                $result['message'],
                []
            ),
            Response::HTTP_OK
        );
    }

    public function changeStatus(Request $request)
    {
        ValidateUUID($request->id);
         try {
        $data = Vendor::where('id', $request->id)->firstOrFail();
        } catch (ModelNotFoundException $e) {
            return response()->json(responseBuilder(Response::HTTP_NOT_FOUND, "Vendor type not found.", []), Response::HTTP_NOT_FOUND);
        }
        
        $this->vendorService->updateStatus($request->id,$data->row_status);
        return response()->json(responseBuilder(Response::HTTP_OK, "Status changed successfully.", []), Response::HTTP_OK);
    }

  

public function export(Request $request): JsonResponse
{
    try {
        $request->validate([
            'type'    => 'required|in:pdf,excel',
            'columns' => 'required|array|min:1',
            'ids'     => 'nullable|array'
        ]);

        $ids     = $request->input('ids', []);
        $type    = $request->input('type');
        $columns = $request->input('columns', []);

        $validColumns = [
            'name',
            'vendor_type_id',
            'contact_number',
            'contact_person',
            'email_address',
            'physical_address',
            'additional_notes',
            'row_status'
        ];

        $columns = array_values(array_unique(array_intersect($columns, $validColumns)));

        if (empty($columns)) {
            return response()->json([
                'success' => false,
                'message' => 'No valid columns selected for export',
            ], 422);
        }

        $fileName = 'vendor-' . now()->format('Y-m-d_H-i-s');
        $export   = new VendorExport($ids, $columns);

        if ($type === 'pdf') {
            $headings = $export->headings();
            $rows     = $export->collection()
                               ->map(function ($row) use ($export) {
                                   return $export->map($row);
                               })
                               ->toArray();

            $colCount = count($headings);

            if ($colCount <= 3) {
                $fontSize = '9px';
            } elseif ($colCount <= 5) {
                $fontSize = '10px';
            } elseif ($colCount <= 7) {
                $fontSize = '10px';
            } else {
                $fontSize = '10px';
            }

            $html = '<!DOCTYPE html>
<html>
<head>
<style>
    @page { margin: 6mm; size: A4 landscape; }
    body { font-family: DejaVu Sans, sans-serif; font-size: ' . $fontSize . '; margin: 0; }
    table { width: 100%; border-collapse: collapse; table-layout: fixed; }
    th {
        background: #e0e0e0;
        font-weight: bold;
        text-align: center;
        padding: 3px 1px;
        border: 1px solid #000000;
        word-wrap: break-word;
        overflow-wrap: break-word;
    }
    td {
        text-align: center;
        padding: 3px 1px;
        border: 1px solid #000000;
        word-wrap: break-word;
        overflow-wrap: break-word;
    }
    tr:nth-child(even) { background: #ffffff; }
</style>
</head>
<body>
<table>
<thead><tr>';

            foreach ($headings as $heading) {
                $html .= '<th>' . htmlspecialchars($heading) . '</th>';
            }

            $html .= '</tr></thead><tbody>';

            foreach ($rows as $row) {
                $html .= '<tr>';
                foreach ($row as $cell) {
                    $html .= '<td>' . htmlspecialchars((string) $cell) . '</td>';
                }
                $html .= '</tr>';
            }

            $html .= '</tbody></table></body></html>';

            $pdf = \Barryvdh\DomPDF\Facade\Pdf::loadHTML($html)
                       ->setPaper('A4', 'landscape');

            return response()->json([
                'status_code' => 200,
                'message'     => 'Success',
                'data'        => [
                    'file'      => base64_encode($pdf->output()),
                    'file_name' => $fileName . '.pdf',
                ]
            ]);
        }

        $fileContent = Excel::raw($export, \Maatwebsite\Excel\Excel::XLSX);

        return response()->json([
            'status_code' => 200,
            'message'     => 'Success',
            'data'        => [
                'file'      => base64_encode($fileContent),
                'file_name' => $fileName . '.xlsx',
            ]
        ]);

    } catch (\Exception $e) {
        \Log::error('Vendor Export Error: ' . $e->getMessage());
        return response()->json([
            'success' => false,
            'message' => 'Failed to generate export: ' . $e->getMessage(),
        ], 500);
    }
}
}
